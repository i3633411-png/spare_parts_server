const express = require('express');
const cors = require('cors');
const axios = require('axios');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(cors());
app.use(express.json());

// قاعدة بيانات وهمية لتخزين الطلبات والعروض
let orders = [];

// مسار استقبال طلبات العملاء
app.post('/api/orders', (req, res) => {
  const newOrder = {
    id: Date.now().toString(),
    ...req.body,
    status: 'بانتظار العروض',
    price: null,
    qualityType: null,
    qualityGrade: null,
    merchantNotes: null,
    createdAt: new Date().toISOString()
  };
  orders.push(newOrder);
  res.status(201).json({ success: true, data: newOrder });
});

// مسار جلب جميع الطلبات
app.get('/api/orders', (req, res) => {
  res.json(orders);
});

// مسار استقبال عرض السعر والجودة والملاحظات من التاجر
app.post('/api/orders/:id/offer', (req, res) => {
  const orderId = req.params.id;
  const { price, qualityType, qualityGrade, notes } = req.body;

  const orderIndex = orders.findIndex(o => o.id.toString() === orderId.toString());
  if (orderIndex !== -1) {
    orders[orderIndex].price = price;
    orders[orderIndex].qualityType = qualityType; // أصلي أو تجاري
    orders[orderIndex].qualityGrade = qualityGrade; // درجة أولى، ثانية...
    orders[orderIndex].merchantNotes = notes;
    orders[orderIndex].status = 'تم إرسال العرض';

    res.json({ success: true, data: orders[orderIndex] });
  } else {
    res.status(404).json({ success: false, message: 'الطلب غير موجود' });
  }
});

// مسار فك رقم الهيكل (VIN Decoder) عبر الـ API المجاني
app.get('/api/decode-vin/:vin', async (req, res) => {
  const vin = req.params.vin;
  try {
    const response = await axios.get(`https://vpic.nhtsa.dot.gov/api/vehicles/decodevinvalues/${vin}?format=json`);
    const results = response.data.Results;
    if (results && results.length > 0) {
      const carData = results[0];
      res.json({
        success: true,
        data: {
          make: carData.Make || 'غير معروف',
          model: carData.Model || 'غير معروف',
          year: carData.ModelYear || 'غير معروف'
        }
      });
    } else {
      res.status(404).json({ success: false, message: 'لم يتم العثور على بيانات رقم الهيكل' });
    }
  } catch (error) {
    res.status(500).json({ success: false, message: 'خطأ في الاتصال بخدمة فك رقم الهيكل' });
  }
});
